============================== FUNCIONARIO ==============================
CREATE TABLE Cargo (
    id_cargo SERIAL,
    nome VARCHAR(50),

    CONSTRAINT pk_cargo PRIMARY KEY (id_cargo),
    CONSTRAINT cargo_nome_nn CHECK (nome IS NOT NULL),
    CONSTRAINT un_cargo_nome UNIQUE (nome)
);

CREATE TABLE Funcionario (
    id_funcionario SERIAL,
    nome VARCHAR(100),
    cpf CHAR(11),
    telefone VARCHAR(15),
    nis CHAR(11),
    nascimento DATE,
    sexo CHAR(1),
    endereco VARCHAR(200),
    id_cargo INTEGER,

    CONSTRAINT pk_funcionario PRIMARY KEY (id_funcionario),

    CONSTRAINT func_nome CHECK (nome IS NOT NULL),
    CONSTRAINT func_nascimento CHECK (nascimento IS NOT NULL),
    CONSTRAINT func_sexo CHECK (sexo IS NOT NULL),

    CONSTRAINT func_cpf_nn CHECK (cpf IS NOT NULL),
    CONSTRAINT func_nis_nn CHECK (nis IS NOT NULL),

    CONSTRAINT un_cpf UNIQUE (cpf),
    CONSTRAINT un_nis UNIQUE (nis),

    CONSTRAINT chk_sexo CHECK (sexo IN ('M', 'F', 'O')),

    CONSTRAINT chk_cpf_format CHECK (cpf ~ '^[0-9]{11}$'),

    CONSTRAINT chk_telefone_format CHECK (telefone ~ '^[0-9]+$'),

    CONSTRAINT fk_funcionario_cargo 
        FOREIGN KEY (id_cargo) REFERENCES Cargo(id_cargo)
);

============================== FAMILIAR ==============================
CREATE TABLE Familiar (
    id_familiar SERIAL,
    nome VARCHAR(100),
    parentesco VARCHAR(50),
    profissao VARCHAR(100),
    renda NUMERIC(10,2),
    telefone VARCHAR(15),

    CONSTRAINT pk_familiar PRIMARY KEY (id_familiar),

    CONSTRAINT fam_nome_nn CHECK (nome IS NOT NULL),
    CONSTRAINT fam_parentesco_nn CHECK (parentesco IS NOT NULL),

    CONSTRAINT chk_renda_valida CHECK (renda >= 0),

    CONSTRAINT chk_telefone_format CHECK (telefone ~ '^[0-9]+$')
);

CREATE TABLE IF NOT EXISTS Itens_Evento(
	id_evento int,
	id_estoque int,
	qtd int not  null,
	CONSTRAINT PK_ITENS_EVENTO PRIMARY KEY (id_evento,id_estoque),
	CONSTRAINT FK_ITENS_EVENTO_Estoque FOREIGN KEY (id_estoque) REFERENCES ESTOQUE(id),
	CONSTRAINT FK_ITENS_EVENTO_Evento FOREIGN KEY (id_evento) REFERENCES Evento(id)
);

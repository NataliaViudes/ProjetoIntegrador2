-- Tabela ALIMENTO
CREATE TABLE IF NOT EXISTS ALIMENTO (
    id SERIAL,
    nome varchar(50),
    tipo varchar(50),
    descricao varchar(200),
    CONSTRAINT PK_ALIMENTO PRIMARY KEY (id)
);

-- Tabela TIPO_ESTOQUE
CREATE TABLE IF NOT EXISTS TIPO_ESTOQUE (
    id SERIAL,
    tipo varchar(50) NOT NULL,
    CONSTRAINT PK_TIPOESTOQUE PRIMARY KEY (id),
    CONSTRAINT UN_TIPO_TIPOESTOQUE UNIQUE (tipo)
);

-- Tabela ESTOQUE
CREATE TABLE IF NOT EXISTS ESTOQUE (
    id SERIAL,
    descricao varchar(200),
    qtd int,
    id_tipo_estoque int,
    CONSTRAINT PK_ESTOQUE PRIMARY KEY (id),
    CONSTRAINT FK_ESTOQUE_TIPO_ESTOQUE FOREIGN KEY (id_tipo_estoque) REFERENCES TIPO_ESTOQUE(id)
);

CREATE TABLE IF NOT EXISTS alimento (
    id SERIAL,
    nome varchar(50),
    tipo varchar(50),
    descricao varchar(200),
    CONSTRAINT PK_ALIMENTO PRIMARY KEY (id)
);

CREATE TABLE cardapio (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    data DATE NOT NULL,
    hora TIME NOT NULL,
    id_agendamento INTEGER,

    CONSTRAINT fk_cardapio_agendamento
        FOREIGN KEY (id_agendamento)
        REFERENCES agendamento_atividade(id)
        ON DELETE SET NULL
);

CREATE TABLE itens_cardapio (
    id_cardapio INTEGER NOT NULL,
    id_alimento INTEGER NOT NULL,
    quantidade INTEGER NOT NULL,

    PRIMARY KEY (id_cardapio, id_alimento),

    CONSTRAINT fk_itens_cardapio_cardapio
        FOREIGN KEY (id_cardapio)
        REFERENCES cardapio(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_itens_cardapio_alimento
        FOREIGN KEY (id_alimento)
        REFERENCES alimento(id)
        ON DELETE CASCADE
);

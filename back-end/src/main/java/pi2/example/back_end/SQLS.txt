CREATE TABLE categoria (
    id_categoria SERIAL,
    nome VARCHAR(100) NOT NULL,

    CONSTRAINT pk_categoria PRIMARY KEY (id_categoria),
    CONSTRAINT un_categoria_nome UNIQUE (nome)
);

CREATE TABLE Cargo (
    id_cargo SERIAL,
    nome VARCHAR(50) NOT NULL, -- alterado para NOT NULL

    CONSTRAINT pk_cargo PRIMARY KEY (id_cargo),
    CONSTRAINT un_cargo_nome UNIQUE (nome)
);

CREATE TABLE Funcionario (
    id_funcionario SERIAL,
    nome VARCHAR(100),
    cpf CHAR(11),
    telefone VARCHAR(15),
    nis CHAR(11),
    nascimento DATE,
    sexo CHAR(1),
    endereco VARCHAR(200),
    id_cargo INTEGER,

    CONSTRAINT pk_funcionario PRIMARY KEY (id_funcionario),

    CONSTRAINT func_nome CHECK (nome IS NOT NULL),
    CONSTRAINT func_nascimento CHECK (nascimento IS NOT NULL),
    CONSTRAINT func_sexo CHECK (sexo IS NOT NULL),

    CONSTRAINT func_cpf_nn CHECK (cpf IS NOT NULL),
    CONSTRAINT func_nis_nn CHECK (nis IS NOT NULL),

    CONSTRAINT un_cpf UNIQUE (cpf),
    CONSTRAINT un_nis UNIQUE (nis),

    CONSTRAINT chk_sexo CHECK (sexo IN ('M', 'F', 'O')),

    CONSTRAINT chk_cpf_format CHECK (cpf ~ '^[0-9]{11}$'),

    CONSTRAINT chk_telefone_format CHECK (telefone ~ '^[0-9]+$'),

    CONSTRAINT fk_funcionario_cargo
        FOREIGN KEY (id_cargo) REFERENCES Cargo(id_cargo)
);

CREATE TABLE beneficiario (
    id SERIAL,
    nome VARCHAR(100) NOT NULL,
    nascimento DATE,
    idade INTEGER,
    rg VARCHAR(20),
    cpf CHAR(11),
    nis CHAR(11),
    renda NUMERIC(10,2),
    endereco VARCHAR(200),
    bairro VARCHAR(100),
    tipo_residencia VARCHAR(50),
    telefone VARCHAR(15),
    celular VARCHAR(15),
    celular_recado VARCHAR(15),

    CONSTRAINT pk_beneficiario PRIMARY KEY (id),
    CONSTRAINT un_beneficiario_cpf UNIQUE (cpf),
    CONSTRAINT un_beneficiario_nis UNIQUE (nis),

    CONSTRAINT chk_beneficiario_cpf_format
        CHECK (cpf IS NULL OR cpf ~ '^[0-9]{11}$'),

    CONSTRAINT chk_beneficiario_nis_format
        CHECK (nis IS NULL OR nis ~ '^[0-9]{11}$'),

    CONSTRAINT chk_beneficiario_telefone_format
        CHECK (telefone IS NULL OR telefone ~ '^[0-9]+$'),

    CONSTRAINT chk_beneficiario_celular_format
        CHECK (celular IS NULL OR celular ~ '^[0-9]+$'),

    CONSTRAINT chk_beneficiario_celular_recado_format
        CHECK (celular_recado IS NULL OR celular_recado ~ '^[0-9]+$')
);

CREATE TABLE auxilio (
    id SERIAL,
    descricao TEXT NOT NULL,
    id_beneficiario INTEGER NOT NULL,
    id_categoria INTEGER NOT NULL,

    CONSTRAINT pk_auxilio PRIMARY KEY (id),

    CONSTRAINT fk_auxilio_beneficiario
        FOREIGN KEY (id_beneficiario)
        REFERENCES beneficiario(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_auxilio_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria_auxilio(id)
);

CREATE TABLE Familiar (
    id_familiar SERIAL,
    nome VARCHAR(100),
    parentesco VARCHAR(50),
    profissao VARCHAR(100),
    renda NUMERIC(10,2),
    telefone VARCHAR(15),

    CONSTRAINT pk_familiar PRIMARY KEY (id_familiar),

    CONSTRAINT fam_nome_nn CHECK (nome IS NOT NULL),
    CONSTRAINT fam_parentesco_nn CHECK (parentesco IS NOT NULL),

    CONSTRAINT chk_renda_valida CHECK (renda >= 0),

    CONSTRAINT chk_telefone_format CHECK (telefone ~ '^[0-9]+$')
);

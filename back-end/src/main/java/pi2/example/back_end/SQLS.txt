CREATE TABLE Funcionario (
    id_funcionario SERIAL,
    nome VARCHAR(100),
    cpf CHAR(11),
    telefone VARCHAR(15),
    nis CHAR(11),
    nascimento DATE,
    sexo CHAR(1),
    endereco VARCHAR(200),
    id_cargo INTEGER,

    CONSTRAINT pk_funcionario PRIMARY KEY (id_funcionario),

    CONSTRAINT func_nome CHECK (nome IS NOT NULL),
    CONSTRAINT func_nascimento CHECK (nascimento IS NOT NULL),
    CONSTRAINT func_sexo CHECK (sexo IS NOT NULL),

    CONSTRAINT func_cpf_nn CHECK (cpf IS NOT NULL),
    CONSTRAINT func_nis_nn CHECK (nis IS NOT NULL),

    CONSTRAINT un_cpf UNIQUE (cpf),
    CONSTRAINT un_nis UNIQUE (nis),

    CONSTRAINT chk_sexo CHECK (sexo IN ('M', 'F', 'O')),

    CONSTRAINT chk_cpf_format CHECK (cpf ~ '^[0-9]{11}$'),

    CONSTRAINT chk_telefone_format CHECK (telefone ~ '^[0-9]+$'),

    CONSTRAINT fk_funcionario_cargo
        FOREIGN KEY (id_cargo) REFERENCES Cargo(id_cargo)
);

CREATE TABLE itens_evento (

    id_evento INT NOT NULL,
    id_estoque INT NOT NULL,
    qtd INT NOT NULL,

    CONSTRAINT pk_itens_evento
        PRIMARY KEY (id_evento, id_estoque),

    CONSTRAINT fk_itens_evento_evento
        FOREIGN KEY (id_evento)
        REFERENCES evento(id_evento),

    CONSTRAINT fk_itens_evento_estoque
        FOREIGN KEY (id_estoque)
        REFERENCES estoque(id),

    CONSTRAINT chk_itens_evento_qtd
        CHECK (qtd > 0)
);
CREATE OR REPLACE FUNCTION fn_controlar_estoque_itens_evento()
RETURNS TRIGGER
AS $$
DECLARE
    qtd_disponivel INT;
    diferenca INT;
BEGIN

    -- =====================================================
    -- INSERT
    -- =====================================================

    IF TG_OP = 'INSERT' THEN

        SELECT qtd
        INTO qtd_disponivel
        FROM estoque
        WHERE id = NEW.id_estoque;

        IF qtd_disponivel < NEW.qtd THEN
            RAISE EXCEPTION
            'Quantidade insuficiente no estoque';
        END IF;

        UPDATE estoque
        SET qtd = qtd - NEW.qtd
        WHERE id = NEW.id_estoque;

        RETURN NEW;
    END IF;

    -- =====================================================
    -- DELETE
    -- =====================================================

    IF TG_OP = 'DELETE' THEN

        UPDATE estoque
        SET qtd = qtd + OLD.qtd
        WHERE id = OLD.id_estoque;

        RETURN OLD;
    END IF;

    -- =====================================================
    -- UPDATE
    -- =====================================================

    IF TG_OP = 'UPDATE' THEN

        -- =============================================
        -- MESMO ESTOQUE
        -- =============================================

        IF NEW.id_estoque = OLD.id_estoque THEN

            diferenca := NEW.qtd - OLD.qtd;

            -- AUMENTOU
            IF diferenca > 0 THEN

                SELECT qtd
                INTO qtd_disponivel
                FROM estoque
                WHERE id = NEW.id_estoque;

                IF qtd_disponivel < diferenca THEN
                    RAISE EXCEPTION
                    'Quantidade insuficiente no estoque';
                END IF;

                UPDATE estoque
                SET qtd = qtd - diferenca
                WHERE id = NEW.id_estoque;

            END IF;

            -- DIMINUIU
            IF diferenca < 0 THEN

                UPDATE estoque
                SET qtd = qtd + ABS(diferenca)
                WHERE id = NEW.id_estoque;

            END IF;

        END IF;

        RETURN NEW;

    END IF;

    RETURN NULL;

END;
$$
LANGUAGE plpgsql;

CREATE TRIGGER trg_controlar_estoque_itens_evento
BEFORE INSERT OR UPDATE OR DELETE
ON itens_evento
FOR EACH ROW
EXECUTE FUNCTION fn_controlar_estoque_itens_evento();

CREATE TABLE IF NOT EXISTS itens_evento (

    id_evento INT NOT NULL,
    id_estoque INT NOT NULL,
    qtd INT NOT NULL,

    CONSTRAINT pk_itens_evento
        PRIMARY KEY (id_evento, id_estoque),

    CONSTRAINT fk_itens_evento_evento
        FOREIGN KEY (id_evento)
        REFERENCES evento(id_evento),

    CONSTRAINT fk_itens_evento_estoque
        FOREIGN KEY (id_estoque)
        REFERENCES estoque(id),

    CONSTRAINT chk_itens_evento_qtd
        CHECK (qtd > 0)
);
CREATE OR REPLACE FUNCTION fn_controlar_estoque_itens_evento()
RETURNS TRIGGER
AS $$
DECLARE
    qtd_disponivel INT;
    diferenca INT;
BEGIN

    -- =====================================================
    -- INSERT
    -- =====================================================

    IF TG_OP = 'INSERT' THEN

        SELECT qtd
        INTO qtd_disponivel
        FROM estoque
        WHERE id = NEW.id_estoque;

        IF qtd_disponivel < NEW.qtd THEN
            RAISE EXCEPTION
            'Quantidade insuficiente no estoque';
        END IF;

        UPDATE estoque
        SET qtd = qtd - NEW.qtd
        WHERE id = NEW.id_estoque;

        RETURN NEW;
    END IF;

    -- =====================================================
    -- DELETE
    -- =====================================================

    IF TG_OP = 'DELETE' THEN

        UPDATE estoque
        SET qtd = qtd + OLD.qtd
        WHERE id = OLD.id_estoque;

        RETURN OLD;
    END IF;

    -- =====================================================
    -- UPDATE
    -- =====================================================

    IF TG_OP = 'UPDATE' THEN

        -- =============================================
        -- MESMO ESTOQUE
        -- =============================================

        IF NEW.id_estoque = OLD.id_estoque THEN

            diferenca := NEW.qtd - OLD.qtd;

            -- AUMENTOU
            IF diferenca > 0 THEN

                SELECT qtd
                INTO qtd_disponivel
                FROM estoque
                WHERE id = NEW.id_estoque;

                IF qtd_disponivel < diferenca THEN
                    RAISE EXCEPTION
                    'Quantidade insuficiente no estoque';
                END IF;

                UPDATE estoque
                SET qtd = qtd - diferenca
                WHERE id = NEW.id_estoque;

            END IF;

            -- DIMINUIU
            IF diferenca < 0 THEN

                UPDATE estoque
                SET qtd = qtd + ABS(diferenca)
                WHERE id = NEW.id_estoque;

            END IF;

        END IF;

        RETURN NEW;

    END IF;

    RETURN NULL;

END;
$$
LANGUAGE plpgsql;

CREATE TRIGGER trg_controlar_estoque_itens_evento
BEFORE INSERT OR UPDATE OR DELETE
ON itens_evento
FOR EACH ROW
EXECUTE FUNCTION fn_controlar_estoque_itens_evento();

        FOREIGN KEY (id_agendamento)
        REFERENCES agendamento_atividade(id)

        FOREIGN KEY (id_beneficiario)
        REFERENCES beneficiario(id)
        ON DELETE SET NULL,

    CONSTRAINT chk_tipo_ocorrencia
        CHECK (tipo IN ('GERAL', 'INDIVIDUAL')),

    CONSTRAINT chk_ocorrencia_tipo
        CHECK (
            (tipo = 'GERAL' AND id_beneficiario IS NULL)
            OR
            (tipo = 'INDIVIDUAL' AND id_beneficiario IS NOT NULL)
        )
);

CREATE TABLE IF NOT EXISTS prescricao (
    id SERIAL,

    dosagem VARCHAR(100) NOT NULL,
    quantidade INTEGER NOT NULL,

    horario TIMESTAMP NOT NULL,

    intervalo VARCHAR(20) NOT NULL,

    id_beneficiario INTEGER NOT NULL,
    id_remedio INTEGER NOT NULL,

    CONSTRAINT pk_prescricao
        PRIMARY KEY (id),

    CONSTRAINT fk_prescricao_beneficiario
        FOREIGN KEY (id_beneficiario)
        REFERENCES beneficiario(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_prescricao_remedio
        FOREIGN KEY (id_remedio)
        REFERENCES remedio(id)
        ON DELETE CASCADE,

    CONSTRAINT chk_prescricao_quantidade
        CHECK (quantidade > 0),

    CONSTRAINT chk_prescricao_intervalo
        CHECK (
            intervalo IN (
                '6h',
                '8h',
                '12h',
                '1dia',
                '1semana'
            )
        )
);
